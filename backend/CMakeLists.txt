cmake_minimum_required(VERSION 3.20)
project(mini-ncu LANGUAGES CXX)
add_executable(mini-ncu src/main.cpp)
target_compile_features(mini-ncu PRIVATE cxx_std_20)
target_include_directories(mini-ncu PRIVATE include)
target_compile_options(mini-ncu PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(NCU_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(NCU_SANITIZE)
    target_compile_options(mini-ncu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mini-ncu PRIVATE -fsanitize=address,undefined)
endif()
